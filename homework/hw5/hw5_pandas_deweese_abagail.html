with open("../../homework/hw5/hw5_raw.txt", "r", encoding="utf-8") as f:
    fin_lines = f.readlines()

while not fin_lines[-1].startswith("["):
    fin_lines = fin_lines[:-1]
fin_lines = fin_lines[:-1]

fin_text = "".join(fin_lines)
Code 
songs = re.split(r"\*[^*]*\*", fin_text)
Code 
rows = []
songs_num = 0

for song in songs:
    songs_num += 1
    lines = re.split(r"[\n]+", song.lower())   # split on \W AND on _
    lines = [l for l in lines if l != ""]
    for l in lines:
        rows.append({"line": l, "songs": songs_num})

fin_df = pd.DataFrame(rows)
Code
